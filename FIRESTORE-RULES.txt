rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {
    match /flashOffers/{offerId} {
      allow read: if true;
      allow create, update, delete: if request.auth != null
        && request.auth.token.email == "REEMPLAZA_CON_TU_EMAIL_ADMIN";
    }
    match /flashSettings/{docId} {
      allow read: if true;
      allow create, update, delete: if request.auth != null
        && request.auth.token.email == "REEMPLAZA_CON_TU_EMAIL_ADMIN";
    }
  }
}
